# Documentation site toolchain (MkDocs + Material theme). Pinned so local
# builds and the CI/deploy workflow (.github/workflows/docs.yml) are
# reproducible. Install into a virtualenv:
#
#   python3 -m venv .venv-docs
#   .venv-docs/bin/pip install -r requirements-docs.txt
#   .venv-docs/bin/mkdocs serve      # live preview at http://127.0.0.1:8000
#   .venv-docs/bin/mkdocs build --strict
mkdocs==1.6.1
# mkdocs-material: 9.6.x capped pymdown-extensions at ~=10.x; 9.7.x relaxes the
# requirement to >=10.2, which the security pin below needs.

# fixed in 11.0.1). Earlier advisories (GHSA-62q4-447f-wv8h, GHSA-r6h4-mm7h-8pmq)
# stay cleared.
pymdown-extensions==11.0.1
# Pygments: 2.20.0 clears the 2.18.x GUID-regex ReDoS and renders cleanly. (The
# older anchor_linenums crash on 2.19+ was a pymdownx bug, fixed by the
# pymdown-extensions bump above — the two must move together.)
pygments==2.20.0
# mkdocs-redirects: emits a stub page at every URL the #3855 reorganisation
